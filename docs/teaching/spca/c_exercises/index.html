String to Integer#

Given a string of the form [0-9+], write a function that returns this as an integer, without using any library functions. Assume all inputs are valid.

int str_to_int(const char* s);

Show Solution 
int char_to_int(char c) {
    // '0' has ascii value 48, '2' 49, ... '9' 57
    return c - 48;
}

int str_length(const char *c) {
    int i = 0; 
    while (c[i] != '\0') {
        i++;
    }
    return i;
}

int str_to_int(const char* s) {
    int str_len = str_length(s);
    int result = 0; 
    int base = 10;
    int cur_exponent = 1; 

    for (int i = str_len - 1; i >= 0; i--) {
        result += s[i] * cur_exponent;
        cur_exponent *= base;
    }
    return result;
}

IP Address Parsing#

Given the string representation of an IPv4 Address (eg. 192.168.0.1), return the IP address as a uint32_t where byte 1 = 192, byte 2 = 128, byte 3 = 0, byte 4 = 1. Assume all inputs are valid.

typedef uint32_t IPv4;
IPv4 str_to_ipv4(const char* ip)

Show Solution 
typedef uint32_t IPv4;

// copies n bytes from dst to src
void copy_n(char *dst, const char *src, int n) {
    for (int i = 0; i < n; i++) {
        dst[i] = src[i];
    }
}

IPv4 str_to_ipv4(const char* ip) {
    IPv4 parsed_ip;
    int len = str_length(ip);

    int cur_start = 0;
    int cur_block = 0;

    for (int i = 0; i < len; i++) {
        if (ip[i] == '.') {
            char[3] cur_block = '0';
            copy_n(cur_block, ip[i], i - cur_start);
            parsed_ip[cur_block++] = str_to_int(cur_block);
            cur_start = i + 1;
        }
    }
    return parsed_ip;
}

Routing Table#

Given a parsed IP address, implement a routing table. Assume all inputs are valid


typedef struct {
    // implement internals
} RoutingTable;

void routing_table_init(RoutingTable *rt, IPv4 default_route);
void add_route(RoutingTable *rt, IPv4 subnet, uint32_t subnet_size, IPv4 destination);
IPv4 get_route(RoutingTable *rt, const char* ip);

Show Solution 
struct Node {
    struct Node *zero;
    struct Node *one;
    IPv4 route;
    bool has_route;
};

typedef struct {
    struct Node *root;
} RoutingTable;

void routing_table_init(RoutingTable *rt, IPv4 default_route) {
    rt->root = calloc(1, sizeof(struct Node));
    rt->root->has_route = true;
    rt->root->route = default_route;
}

IPv4 get_route(RoutingTable *rt, const char* destination) {

    struct Node cur_node = rt->root;
    IPv4 best_route = cur_node->route;

    for (int i = 0; i < 32; i++) {
        
        int cur_bit = (destination >> i) & 0x01;
        if (cur_bit == 1) {
            cur_node = cur_node->one;
        } else {
            cur_node = cur_node->zero;
        }

        // update best route to longest prefix
        if (cur_node->has_route) {
            best_route = cur_node->route;
        }
    }
    return best_route;
}

void add_route(RoutingTable *rt, IPv4 subnet, uint32_t subnet_size, IPv4 destination) {

    struct Node *cur_node = rt->root;
    for (int i = 0; i < subnet_size; i++) {
        int cur_bit = (subnet >> i) & 0x1;
        if (cur_bit == 1) {
            if (cur_node->one) {
                cur_node = cur_node->one;
            } else {
                cur_node->one = calloc(1, sizeof(struct Node));
                cur_node = cur_node->one;
            }
        } else {
            if (cur_node->zero) {
                cur_node = cur_node->zero;
            } else {
                cur_node->zero = calloc(1, sizeof(struct Node));
                cur_node = cur_node->zero;
            }
        }
    }
    cur_node->route = destination;
    cur_node->has_route = true;
}

struct Node {
    struct Node *prev;
    struct Node *next;
    int value;
};

typedef struct {
    int size;
    struct Node *head;
    struct Node *tail;
} List;

void list_push_back(List *l, int v) {
    struct Node n = malloc(sizeof(struct Node));
    n->prev = l->tail;
    n->next = NULL;
    n->value = value;

    if (l->size == 0) {
        l->tail = n;
        l->head = n;
    } else {
        l->tail = n;
    }
    l->size++;
}

void list_push_front(List *l, int v) {
    struct Node n = malloc(sizeof(struct Node));
    n->prev = NULL; 
    n->next = l->head;

    if (l->size == 0) {
        l->tail = n;
        l->head = n;
    } else {
        l->head = n;
    }
    l->size++;

}

int list_size(List *l) {
    return l->size;
}

// internal helper
struct Node *list_get_node(List *l, int idx) {
    struct Node *cur = l->head;
    for (int i = 0; i < n; i++) {
        cur = cur->next;
    }
    return cur;
}

int list_get(List *l, int idx) {
    struct Node *n = list_get_node(l, idx);
    return n->value;
}

void list_erase(List *l, int idx) {
    struct Node *n = list_get_node(l, idx);
    struct Node *prev = n->prev;
    struct node *next = n->next;

    prev->next = n->next;
    next->prev = prev;
    free(n);
}

void hashmap_init(HashMap *m, int capacity, int max_load_factor, size_t (hasher)(int)) {
    m->capacity = capacity;
    m->hasher = hasher;
    m->max_load_factor = max_load_factor;
    m->buckets = calloc(m->capacity, sizeof(List));
    m->size = 0;
}

#define LIST_FOR_EACH(l, n) 
    for (struct Node *(n) = (l)->head; (n) != NULL; (n) = (n)->next)
    
void hashmap_resize(HashMap *m, int new_capacity) {
    List *buckets_old = m->buckets;
    unsigned int capacity_old = m->capacity;

    // resize buckets, reinset values
    List *buckets_new = calloc(m->capacity << 2, sizeof(List));
    m->buckets = buckets_new;
    m->capacity = new_capacity;

    for (int i = 0; i < capacity_old; i++) {
        LIST_FOR_EACH(buckets_old, cur_node) {
            // add helper functions
            size_t hash = m->hasher(cur_node->value);
            struct Node *head = m->buckets[hash % m->capacity];
            cur_node->prev = NULL;
            cur_node->next = head;       
        }
    }
    free(buckets_old);
}

int hashmap_get(HashMap *m, int k) {
     size_t bucket_idx = m->hasher(k) %  m->n_buckets;
     struct Node *cur = m->buckets[bucket_idx];
     
     struct Node *n = hashmap_get_node(m, k);
     return n->value;
}

bool hashmap_contains(HashMap *m, int k) {
    return hashmap_get_node(m, k) != NULL;
}

struct Node *hashmap_get_node(HashMap *m, int k) {
    size_t bucket_idx = m->hasher(k) %  m->n_buckets;
    struct Node *cur = m->buckets[bucket_idx];

    while (cur) {
            if (cur->key == k) {
                return cur;
            }
            cur = cur->next;
         }
         return NULL;
}

void hashmap_put(HashMap *m, int k, int v) {
     size_t bucket_idx = m->hasher(k) %  m->n_buckets;
     List* l = &m->buckets[bucket_idx];

    if (hashmap_contains(m, k)) {
        return;
    }
    list_insert(l, v);

    // rehash if necessary
    if (m->size / m->capacity > m->max_load_factor) {
        hashmap_resize(m, m->capacity * 2);
    }
}

void hashmap_erase(HashMap *m, int k) {
    struct List *l = hashmap_get_list(m, k);
    struct Node *n = list_find_node(l, v);
    list_erase(l, n);
}

int hashmap_size(HashMap *m) {
    return m->size;
}

void hashmap_clear(HashMap *m) {
    for (int i = 0; i < m->size; i++) {
        list_clear(&m->buckets[i]);
    }
    m->size = 0;
}
